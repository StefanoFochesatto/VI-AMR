\noindent \textbf{Three algorithms (see Papadapoulos \& Hinterm\"uller, 2026).}

\medskip
\noindent Problem (linear operator $\bF(\bu) = A\bu - \bb$):
\begin{align}
A \bu + \blambda &= \bb  \tag{3.1a} \\
\blambda - \left(\blambda + c(\bu-\bvarphi)\right)_+ &= \bzero \tag{3.1b}
\end{align}

\vspace{-3mm}
\begin{algorithm}[!ht]
\caption{HIK solver for (3.1)}
\label{alg:hik}
\begin{algorithmic}[1]
\numberfromzero
\State \textbf{parameters:} $c>0$, $\tol>0$.  \hfill {\scriptsize (\emph{HIK: $c=1$.})}
\State \textbf{input:} $A$, $\bb := M\bff$, $\bvarphi$, $\bu^0$.
\State $\blambda^0 = \bzero$, $\Ac{0} = \emptyset$ and $\Ic{0} = \{1:N\}$.
\State Initialize $k = 0$.
\Repeat
  \State Solve $A_{\Ic{k},\Ic{k}}\,\bdelta^k_{\Ic{k}}
         = -A_{\Ic{k},\Ac{k}}(\bvarphi - \bu^k)_{\Ac{k}} - (A\bu^k - \bb)_{\Ic{k}}$.
  \State Assign $\bu^{k+1}_{\Ic{k}} = \bu^k_{\Ic{k}} + \bdelta^k_{\Ic{k}}$
         and $\bu^{k+1}_{\Ac{k}} = \bvarphi_{\Ac{k}}$.
  \State Assign $\blambda^{k+1}_{\Ac{k}} = -(A\bu^{k+1} - \bb)_{\Ac{k}}$
         and $\blambda^{k+1}_{\Ic{k}} = \bzero$.
  \State Assign $\Ac{k+1} = \{i : \lambda^{k+1}_i + c(u^{k+1}_i - \varphi_i) > 0\}$
         and $\Ic{k+1} = \{1:N\}\backslash\Ac{k+1}$.
  \State Assign $k \leftarrow k+1$.
\Until{$\|\blambda^k - (\blambda^k + c(\bu^k - \bvarphi))_+\|_{\ell^2} + \|A\bu^k + \blambda^k - \bb\|_{\ell^2} < \tol$.}
\end{algorithmic}
\end{algorithm}


\medskip
\noindent Problem (nonlinear operator $\bF(\bu)$):
\begin{align}
\bF(\bu) + \blambda &= \bzero  \tag{99a} \\
\blambda - \left(\blambda + c(\bu-\bvarphi)\right)_+ &= \bzero \tag{99b}
\end{align}
equivalently  $\left<\bF(\bu),\bv-\bu\right> \ge 0$ for all $\bv \in K = \left\{\bv\le \bvarphi\right\}$.  Let's assume $\bF$ is differentiable, and that $\bF(\bu+\bw) = \bF(\bu) + \bJ(\bu) \bw + o(\|\bw\|)$.

\vspace{-1mm}
\begin{algorithm}[!ht]
\caption{``HIK'' solver for (99)}
\label{alg:hiknl}
\begin{algorithmic}[1]
\numberfromzero
\State \textbf{parameters:} $c>0$, $\tol>0$.  \hfill {\scriptsize (\emph{HIK: $c=1$.})}
\State \textbf{input:} $\bF:\RR^n\to\RR^n$, $\bJ:\RR^n\to\RR^{n\times n}$, $\bvarphi$, $\bu^0$.  \hfill {\scriptsize (\emph{HIK: $\bF(\bu)=A\bu-\bb$ and $\bJ(\bu)=A$.})}
\State $\blambda^0 = \bzero$, $\Ac{0} = \emptyset$ and $\Ic{0} = \{1:N\}$.
\State Initialize $k = 0$.
\Repeat
  \State Solve $\bJ(\bu^k)_{\Ic{k},\Ic{k}}\,\bdelta^k_{\Ic{k}}
         = -\bJ(\bu^k)_{\Ic{k},\Ac{k}}(\bvarphi - \bu^k)_{\Ac{k}} - \bF(\bu^k)_{\Ic{k}}$.
  \State Assign $\bu^{k+1}_{\Ic{k}} = \bu^k_{\Ic{k}} + \bdelta^k_{\Ic{k}}$
         and $\bu^{k+1}_{\Ac{k}} = \bvarphi_{\Ac{k}}$.
  \State Assign $\blambda^{k+1}_{\Ac{k}} = -\bF(\bu^{k+1})_{\Ac{k}}$
         and $\blambda^{k+1}_{\Ic{k}} = \bzero$.
  \State Assign $\Ac{k+1} = \{i : \lambda^{k+1}_i + c(u^{k+1}_i - \varphi_i) > 0\}$
         and $\Ic{k+1} = \{1:N\}\backslash\Ac{k+1}$.
  \State Assign $k \leftarrow k+1$.
\Until{$\|\blambda^k - (\blambda^k + c(\bu^k - \bvarphi))_+\|_{\ell^2} + \|\bF(\bu^k) + \blambda^k\|_{\ell^2} < \tol$.}
\end{algorithmic}
\end{algorithm}

\smallskip
\noindent Define truncation $\pi:\RR^n\to K$.  Alg 3 has admissible iterates and a pre-setting of the sets, thus losing the $\bJ(\bu^k)_{\Ic{k},\Ac{k}}$ term in the solve line.

\vspace{-1mm}
\begin{algorithm}[!ht]
\caption{RSLS solver for (99)}
\label{alg:rsls}
\begin{algorithmic}[1]
\numberfromzero
\State \textbf{parameters:} $\tol>0$.
\State \textbf{input:} $\bF:{\color{red} K}\to\RR^n$, $\bJ:{\color{red} K}\to\RR^{n\times n}$, $\bvarphi$, $\bu^0$.
\State $\bu^0 \leftarrow \pi[\bu^0]$, $\blambda^0 = -\bF(\bu^0)$, $\Ac{0} = {\color{red} \{i : u^0_i = \varphi_i \text{ \& } \lambda^0_i > 0\}}$, $\Ic{0} = \{1:N\}\backslash\Ac{0}$.
\State Initialize $k = 0$.
\Repeat
  \State Solve $\bJ(\bu^k)_{\Ic{k},\Ic{k}}\,\bdelta^k_{\Ic{k}}
         = - \bF(\bu^k)_{\Ic{k}}$.
  \State Assign $\bdelta^k_{\Ac{k}}=\bzero$ and $\bu^{k+1} = {\color{red} \pi[\bu^k + \alpha\bdelta^k]}$.  \hfill {\scriptsize ({\color{red} \emph{$\alpha$ from line search}})}
  \State Assign $\blambda^{k+1} = -\bF(\bu^{k+1})$.
  \State Assign $\Ac{k+1} = {\color{red} \{i : u^{k+1}_i = \varphi_i \text{ \& } \lambda^{k+1}_i > 0 \}}$
         and $\Ic{k+1} = \{1:N\}\backslash\Ac{k+1}$.
  \State Assign $k \leftarrow k+1$.
\Until{${\color{red} \|\hat\bF(\bu^k)\|_{\ell^2}} < \tol$ {\Large $\strut$}}  \hfill {\scriptsize \Big({\color{red} $\hat\bF(\bu)_i=\begin{cases} \bF(\bu)_i \text{ if } u_i<\varphi_i \\ (\bF(\bu)_i)_+ \text{ if } u_i=\varphi_i \end{cases}$}\Big)}
\end{algorithmic}
\end{algorithm}
